fetch("/", {
  method: "POST",
  headers: {
    "Content-Type": "application/json",
  },
  body: JSON.stringify({
    test: "Hi I am the user"
  })
})
.then(resp => resp.json())
.then(data => {
  document.body.innerHTML = `
    <h1>Upload an Image</h1>

    <div id="dropZone">
      Drag and drop an image here<br />
      or click to choose a file
    </div>

    <input type="file" id="fileInput" accept="image/*" style="display:none;" />
    <img id="preview" alt="Preview" style="max-width:300px; display:none; margin-top:20px;" />
    <br />
    <button id="sendBtn">Send</button>

    <div id="result"></div>
  `;

  const dropZone = document.getElementById("dropZone");
  const fileInput = document.getElementById("fileInput");
  const preview = document.getElementById("preview");
  const sendBtn = document.getElementById("sendBtn");
  const result = document.getElementById("result");

  let selectedFile = null;

  function setFile(file) {
    if (!file || !file.type.startsWith("image/")) {
      result.textContent = "Please select a valid image.";
      return;
    }

    selectedFile = file;
    result.textContent = "";

    const reader = new FileReader();
    reader.onload = function(e) {
      preview.src = e.target.result;
      preview.style.display = "block";
    };
    reader.readAsDataURL(file);
  }

  dropZone.addEventListener("click", () => {
    fileInput.click();
  });

  fileInput.addEventListener("change", (event) => {
    const file = event.target.files[0];
    setFile(file);
  });

  dropZone.addEventListener("dragover", (event) => {
    event.preventDefault();
    dropZone.style.background = "#f3f3f3";
  });

  dropZone.addEventListener("dragleave", () => {
    dropZone.style.background = "";
  });

  dropZone.addEventListener("drop", (event) => {
    event.preventDefault();
    dropZone.style.background = "";

    const file = event.dataTransfer.files[0];
    setFile(file);
  });

  sendBtn.addEventListener("click", async () => {
    if (!selectedFile) {
      result.textContent = "No image selected.";
      return;
    }

    const formData = new FormData();
    formData.append("image", selectedFile);

    try {
      const response = await fetch("/process_galaxy_image/", {
        method: "POST",
        body: formData
      });

      const responseData = await response.json();
      result.textContent = "Server response: " + JSON.stringify(responseData);
    } catch (err) {
      console.error(err);
      result.textContent = "Upload failed.";
    }
  });
})
.catch(err => console.error("Error:", err));
